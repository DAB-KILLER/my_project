name: DAST 

  

on: 

  push: 

    branches: 

      - main 

  workflow_dispatch: 

  

jobs: 

  check-staging: 

    runs-on: ubuntu-latest 

    environment: 

      name: staging 

    steps: 

      - name: Check staging is reachable 

        run: curl -fsS "$STAGING_URL" 

        env: 

          STAGING_URL: ${{ vars.STAGING_URL }} 

  

  zap-baseline: 

    runs-on: ubuntu-latest 

    needs: check-staging 

    environment: 

      name: staging 

    steps: 

      - name: Pull OWASP ZAP image 

        run: docker pull zaproxy/zap-stable 

  

      - name: Run OWASP ZAP baseline scan 

        run: | 

          docker run --rm \ 

            -v "${{ github.workspace }}:/zap/wrk/:rw" \ 

            zaproxy/zap-stable \ 

            zap-baseline.py \ 

            -t "$STAGING_URL" \ 

            -r zap-baseline.html 

        env: 

          STAGING_URL: ${{ vars.STAGING_URL }} 

  

      - name: Upload baseline report 

        if: always() 

        uses: actions/upload-artifact@v4 

        with: 

          name: zap-baseline-report 

          path: zap-baseline.html
